### 起動環境(docker)
```yml
services:
  mc:
    image: itzg/minecraft-server:java17
    environment:
      EULA: "TRUE"
      TYPE: FORGE
      VERSION: "1.20.1"
      FORGE_VERSION: "47.4.10"
      MEMORY: "6G"
      ONLINE_MODE: "TRUE"
    ports:
      - "25565:25565"
    volumes:
      - ./minecraft:/data
    restart: unless-stopped
```
